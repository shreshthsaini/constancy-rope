\section{Additional figures}\label{app:figs}

\Cref{fig:qual} shows held-out samples, \cref{fig:checkpoint} the development checkpoint diagnostic, \cref{fig:spectrum} the octave-band spectra, and \cref{fig:papercode} the development paper-versus-code comparison.

\begin{figure}[H]
\centering
\figorplaceholder[0.95\linewidth]{figs/f4_qualitative.pdf}{3.6in}
\caption{\textbf{At $4096^2$ \ntk{} loses fine detail, while the other arms render comparable detail.} Held-out rows 8 to 10, the first three held-out rows, fixed before any output was viewed; seed 0, one sample per arm. Full frames are Lanczos-reduced; native-resolution crops sit at identical coordinates (A: highest-detail tile of the across-arm mean luminance; B: image centre) and are never resampled. Compositions differ between arms, so a crop can show different content. \dype{} shows faint speckle in flat sky on held-out row 8 (one crop checked).}
\label{fig:qual}
\end{figure}

\begin{figure}[H]
\centering
\figorplaceholder[\linewidth]{figs/f5_checkpoint.pdf}{3.26in}
\caption{\textbf{The development comparison shows a lattice on \fluxdev{}, but checkpoint and guidance both change.} (a)~Dev prompt 0 under official \dype{} code (seed 42) and our \dype{} and \segaofficial{} pipelines (seed 0), at guidance 3.5 (\fluxdev) and 4.5 (\krea); A and B are native 192-pixel windows at the same place in every image. (b)~Whitened radial power spectra of luminance: \fluxdev{} peaks at the lattice period and \krea{} does not in these samples. Official-code evidence covers one prompt per checkpoint; the descriptive eight-prompt peak summaries use our pipeline. The figure does not isolate a checkpoint effect.}
\label{fig:checkpoint}
\end{figure}

\begin{figure}[H]
